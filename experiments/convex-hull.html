10 Convex Hull (Divide & Conquer and Graham Scan)

Aim

Given N distinct points in a 2-D plane, to find the Convex Hull (the smallest convex polygon that contains all the points inside or on its boundary) using (A) the Divide and Conquer technique and (B) the Graham Scan algorithm, and to display the hull points in counter-clockwise order.

Algorithm

A. Divide and Conquer (QuickHull)

  1. Find the leftmost point A and the rightmost point B. Both always lie on the hull.
  2. Line A–B divides the points into a lower set (right of A→B) and an upper set (right of B→A).
  3. DIVIDE: for a set and an edge P→Q, find the point C farthest from the line PQ. C is a hull vertex.
  4. Discard the points inside triangle P-C-Q. Split the rest into the points outside P→C and outside C→Q.
  5. CONQUER: recursively find the hull for (P, C) and (C, Q). Stop when a set is empty.
  6. Output A, lower-hull points, B, upper-hull points. This is the counter-clockwise order.

B. Graham Scan

  1. Pick the lowest point (leftmost if tied) as the anchor p0.
  2. Sort all other points by polar angle around p0 (nearer point first if the angle is equal).
  3. If several points have the same angle, keep only the farthest one.
  4. Push the first three points on a stack.
  5. For every next point p: while the last two stack points and p do not make a left (counter-clockwise) turn, pop the stack. Then push p.
  6. The stack now holds the hull in counter-clockwise order.

Orientation test used in both

cross(o,a,b) = (a.x-o.x)(b.y-o.y) - (a.y-o.y)(b.x-o.x) > 0 : left turn, < 0 : right turn, = 0 : collinear

Code

Part A: Divide and Conquer (q10a.c)
#include<stdio.h>

typedef struct { int x, y; } Point;

Point hull[100];                            /* result, filled in counter-clockwise order */
int m=0;

long cross(Point o, Point a, Point b){
    return (long)(a.x-o.x)*(b.y-o.y) - (long)(a.y-o.y)*(b.x-o.x);
}

long dist2(Point a, Point b){
    return (long)(a.x-b.x)*(a.x-b.x) + (long)(a.y-b.y)*(a.y-b.y);
}

/* DIVIDE: among the points lying strictly to the right of P->Q,
   find the farthest point C. C is a hull vertex. It splits the
   problem into two smaller ones: (P,C) and (C,Q). */
void findHull(Point pts[], int n, Point P, Point Q){
    if(n==0) return;

    int c=-1;
    for(int i=0;i<n;i++){
        if(cross(P,Q,pts[i])<0){
            if(c==-1 || cross(P,Q,pts[i])<cross(P,Q,pts[c]) ||
               (cross(P,Q,pts[i])==cross(P,Q,pts[c]) && dist2(P,pts[i])>dist2(P,pts[c])))
                c=i;
        }
    }
    if(c==-1) return;                       /* no point outside edge P->Q */
    Point C=pts[c];

    /* split the remaining points into those outside P->C and outside C->Q */
    Point s1[100], s2[100];
    int n1=0, n2=0;
    for(int i=0;i<n;i++){
        if(cross(P,C,pts[i])<0) s1[n1++]=pts[i];
        else if(cross(C,Q,pts[i])<0) s2[n2++]=pts[i];
    }

    /* CONQUER: solve both halves, C goes in between (keeps CCW order) */
    findHull(s1,n1,P,C);
    hull[m++]=C;
    findHull(s2,n2,C,Q);
}

int main(){
    Point p[]={{0,3},{1,1},{2,2},{4,4},{0,0},{1,2},{3,1},{3,3}};
    int n=sizeof(p)/sizeof(p[0]);

    /* leftmost (A) and rightmost (B) points are always on the hull */
    int a=0, b=0;
    for(int i=1;i<n;i++){
        if(p[i].x<p[a].x || (p[i].x==p[a].x && p[i].y<p[a].y)) a=i;
        if(p[i].x>p[b].x || (p[i].x==p[b].x && p[i].y>p[b].y)) b=i;
    }
    Point A=p[a], B=p[b];

    /* line A-B divides the points into a lower set and an upper set */
    hull[m++]=A;
    findHull(p,n,A,B);                      /* lower hull : A -> B */
    hull[m++]=B;
    findHull(p,n,B,A);                      /* upper hull : B -> A */

    /* start printing from the lowest (then leftmost) point */
    int s=0;
    for(int i=1;i<m;i++)
        if(hull[i].y<hull[s].y || (hull[i].y==hull[s].y && hull[i].x<hull[s].x)) s=i;

    printf("Convex Hull (Divide and Conquer), counter-clockwise:\n");
    for(int i=0;i<m;i++){
        Point q=hull[(s+i)%m];
        printf("(%d, %d)\n",q.x,q.y);
    }
    return 0;
}

Part B: Graham Scan (q10b.c)
#include<stdio.h>
#include<stdlib.h>

typedef struct { int x, y; } Point;
Point p0;                                   /* anchor (lowest) point */

long cross(Point o, Point a, Point b){
    return (long)(a.x-o.x)*(b.y-o.y) - (long)(a.y-o.y)*(b.x-o.x);
}

long dist2(Point a, Point b){
    return (long)(a.x-b.x)*(a.x-b.x) + (long)(a.y-b.y)*(a.y-b.y);
}

/* sort by polar angle around p0 (counter-clockwise); nearer first on ties */
int cmp(const void *p, const void *q){
    Point a=*(Point*)p, b=*(Point*)q;
    long c=cross(p0,a,b);
    if(c==0) return dist2(p0,a)<dist2(p0,b) ? -1 : 1;
    return c>0 ? -1 : 1;
}

int main(){
    Point p[]={{0,3},{1,1},{2,2},{4,4},{0,0},{1,2},{3,1},{3,3}};
    int n=sizeof(p)/sizeof(p[0]);

    /* Step 1: lowest point (leftmost if tie) goes to p[0] */
    int mn=0;
    for(int i=1;i<n;i++)
        if(p[i].y<p[mn].y || (p[i].y==p[mn].y && p[i].x<p[mn].x)) mn=i;
    Point t=p[0]; p[0]=p[mn]; p[mn]=t;
    p0=p[0];

    /* Step 2: sort the remaining points by polar angle around p0 */
    qsort(p+1,n-1,sizeof(Point),cmp);

    /* Step 2b: for equal angles keep only the farthest point */
    int m=1;
    for(int i=1;i<n;i++){
        while(i<n-1 && cross(p0,p[i],p[i+1])==0) i++;
        p[m++]=p[i];
    }

    if(m<3){ printf("Convex hull not possible\n"); return 0; }

    /* Step 3: scan with a stack */
    Point st[200];
    int top=-1;
    st[++top]=p[0]; st[++top]=p[1]; st[++top]=p[2];
    for(int i=3;i<m;i++){
        while(top>=1 && cross(st[top-1],st[top],p[i])<=0) top--;   /* non-left turn: pop */
        st[++top]=p[i];
    }

    printf("Convex Hull (Graham Scan), counter-clockwise:\n");
    for(int i=0;i<=top;i++) printf("(%d, %d)\n",st[i].x,st[i].y);
    return 0;
}

Time Complexity

AlgorithmBestAverageWorst
Divide & Conquer (QuickHull)O(n log n)O(n log n)O(n²) – very unbalanced splits
Graham ScanO(n log n)O(n log n)O(n log n)

In Graham Scan the sort costs O(n log n) and the scan is O(n), because every point is pushed once and popped at most once.

Space Complexity

Divide & Conquer: O(n) for the point subsets (plus O(log n) recursion stack on balanced splits). Graham Scan: O(n) for the stack.

Recurrence Relation

Divide & Conquer (QuickHull): balanced split T(n) = 2T(n/2) + n ⇒ O(n log n). Worst split T(n) = T(n-1) + n ⇒ O(n²).
Graham Scan: T(n) = O(n log n) [sort] + O(n) [scan] ⇒ O(n log n)

void fh(P s[],int n,P a,P b){

    for(int i=0;i<n;i++) if(cr(a,b,s[i])<0 && (c<0||cr(a,b,s[i])<cr(a,b,s[c])||(cr(a,b,s[i])==cr(a,b,s[c])&&d2(a,s[i])>d2(a,s[c])))) c=i;
